EXECUTIVE SUMMARY
================================================================================

The Proof-of-Reading Protocol (PoRP) is a 4-layer behavioral proof-of-work system designed
to prevent Sybil attacks in SempaiHQ's read-to-earn ecosystem while maintaining zero friction
for honest users. This implementation plan provides a comprehensive 10-week roadmap to
deploy all PoRP components across both web (SempaiHeadquarters) and mobile (SempaiHQ_app)
platforms.

TIMELINE: 10 weeks across 4 phases
ESTIMATED EFFORT: ~260 engineering hours
IMPACT: Eliminates bot drainage, protects low-liquidity tokenomics, enables sustainable growth

class ReadingTracker {
  constructor() {
    this.sessionId = null;
    this.startTime = null;
    this.scrollEvents = [];
    this.interactionEvents = [];
    this.deviceHash = this.generateDeviceHash();
  }

  async startSession(novelId, chapterNumber) {
    const response = await fetch('/api/session/start', {
      method: 'POST',
      body: JSON.stringify({
        novelId,
        chapterNumber,
        deviceHash: this.deviceHash
      })
    });
    this.sessionId = response.sessionId;
    this.startTime = Date.now();
    this.setupEventListeners();
  }

  setupEventListeners() {
    // Scroll tracking
    window.addEventListener('scroll', this.handleScroll.bind(this), { passive: true });
    
    // Interaction tracking
    document.addEventListener('click', this.handleInteraction.bind(this));
    document.addEventListener('keypress', this.handleInteraction.bind(this));
    
    // Visibility tracking
    document.addEventListener('visibilitychange', this.handleVisibility.bind(this));
  }

  handleScroll(event) {
    this.scrollEvents.push({
      timestamp: Date.now() - this.startTime,
      scrollY: window.scrollY,
      scrollHeight: document.documentElement.scrollHeight,
      clientHeight: document.documentElement.clientHeight
    });
  }

  handleInteraction(event) {
    this.interactionEvents.push({
      timestamp: Date.now() - this.startTime,
      type: event.type,
      target: event.target.tagName,
      position: { x: event.clientX, y: event.clientY }
    });
  }

  calculateEntropy() {
    // Analyze randomness in scroll patterns and interactions
    const scrollPattern = this.analyzeScrollPattern();
    const interactionPattern = this.analyzeInteractionPattern();
    return (scrollPattern.entropy + interactionPattern.entropy) / 2;
  }

  analyzeScrollPattern() {
    if (this.scrollEvents.length < 2) return { entropy: 0 };
    
    // Calculate variance in scroll velocities
    const velocities = [];
    for (let i = 1; i < this.scrollEvents.length; i++) {
      const timeDiff = this.scrollEvents[i].timestamp - this.scrollEvents[i-1].timestamp;
      const scrollDiff = this.scrollEvents[i].scrollY - this.scrollEvents[i-1].scrollY;
      velocities.push(Math.abs(scrollDiff / timeDiff));
    }
    
    const mean = velocities.reduce((a, b) => a + b, 0) / velocities.length;
    const variance = velocities.reduce((sum, v) => sum + Math.pow(v - mean, 2), 0) / velocities.length;
    
    // Normalize variance to 0-1 scale
    return { entropy: Math.min(1, variance / 1000) };
  }

  analyzeInteractionPattern() {
    if (this.interactionEvents.length === 0) return { entropy: 0 };
    
    // Calculate time intervals between interactions
    const intervals = [];
    for (let i = 1; i < this.interactionEvents.length; i++) {
      intervals.push(this.interactionEvents[i].timestamp - this.interactionEvents[i-1].timestamp);
    }
    
    // Calculate entropy based on interval variance
    const mean = intervals.reduce((a, b) => a + b, 0) / intervals.length;
    const variance = intervals.reduce((sum, interval) => sum + Math.pow(interval - mean, 2), 0) / intervals.length;
    
    return { entropy: Math.min(1, variance / 10000) };
  }

  generateDeviceHash() {
    const canvas = document.createElement('canvas');
    const ctx = canvas.getContext('2d');
    ctx.textBaseline = 'top';
    ctx.font = '14px Arial';
    ctx.fillText('Device fingerprint hash', 2, 2);
    
    const fingerprint = [
      navigator.userAgent,
      navigator.language,
      screen.width + 'x' + screen.height,
      new Date().getTimezoneOffset(),
      canvas.toDataURL()
    ].join('|');
    
    // Simple hash (in production, use crypto.subtle.digest)
    let hash = 0;
    for (let i = 0; i < fingerprint.length; i++) {
      const char = fingerprint.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash; // Convert to 32-bit integer
    }
    
    return Math.abs(hash).toString(16);
  }

  async completeSession() {
    const sessionData = {
      timeOnPage: Date.now() - this.startTime,
      scrollEvents: this.scrollEvents,
      interactionEvents: this.interactionEvents,
      entropyScore: this.calculateEntropy()
    };

    const response = await fetch('/api/session/verify', {
      method: 'POST',
      body: JSON.stringify({
        sessionId: this.sessionId,
        sessionData
      })
    });

    return response.receipt; // Cryptographic receipt
  }
}

import { NextResponse } from 'next/server';
import { supabase } from '@/services/supabase/supabaseClient';
import crypto from 'crypto';

export async function POST(request) {
  try {
    const { sessionId, sessionData } = await request.json();
    
    // Validate session exists and belongs to user
    const session = await validateSession(sessionId);
    if (!session) {
      return NextResponse.json({ error: 'Invalid session' }, { status: 400 });
    }
    
    // Bot detection heuristics
    const botFlags = detectBots(sessionData);
    if (botFlags.length > 0) {
      await flagSession(sessionId, botFlags);
      return NextResponse.json({ error: 'Session flagged for review' }, { status: 403 });
    }
    
    // Validate time-on-page (minimum 20 seconds)
    if (sessionData.timeOnPage < 20000) {
      return NextResponse.json({ error: 'Insufficient reading time' }, { status: 400 });
    }
    
    // Validate entropy score (minimum 0.6)
    if (sessionData.entropyScore < 0.6) {
      return NextResponse.json({ error: 'Low behavioral entropy' }, { status: 400 });
    }
    
    // Update session with completion data
    await supabase
      .from('reading_sessions')
      .update({
        session_data: sessionData,
        entropy_score: sessionData.entropyScore,
        time_on_page_seconds: Math.floor(sessionData.timeOnPage / 1000),
        status: 'completed',
        completed_at: new Date().toISOString()
      })
      .eq('id', sessionId);
    
    // Issue cryptographic receipt
    const receipt = await issueSessionReceipt(session, sessionData);
    
    return NextResponse.json({ receipt });
  } catch (error) {
    console.error('Session verification error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}

function detectBots(sessionData) {
  const flags = [];
  
  // Check for instant page loads
  if (sessionData.timeOnPage < 5000) flags.push('instant_read');
  
  // Check for uniform scroll patterns
  const scrollVariance = calculateScrollVariance(sessionData.scrollEvents);
  if (scrollVariance < 0.1) flags.push('uniform_scroll');
  
  // Check for lack of interactions
  if (sessionData.interactionEvents.length === 0) flags.push('no_interactions');
  
  // Check for linear scroll progression
  if (isLinearScroll(sessionData.scrollEvents)) flags.push('linear_scroll');
  
  return flags;
}

function calculateScrollVariance(scrollEvents) {
  if (scrollEvents.length < 2) return 0;
  
  const velocities = [];
  for (let i = 1; i < scrollEvents.length; i++) {
    const timeDiff = scrollEvents[i].timestamp - scrollEvents[i-1].timestamp;
    const scrollDiff = scrollEvents[i].scrollY - scrollEvents[i-1].scrollY;
    if (timeDiff > 0) {
      velocities.push(Math.abs(scrollDiff / timeDiff));
    }
  }
  
  if (velocities.length === 0) return 0;
  
  const mean = velocities.reduce((a, b) => a + b, 0) / velocities.length;
  const variance = velocities.reduce((sum, v) => sum + Math.pow(v - mean, 2), 0) / velocities.length;
  
  return Math.min(1, variance / 1000);
}

function isLinearScroll(scrollEvents) {
  if (scrollEvents.length < 3) return false;
  
  // Check if scroll positions increase consistently
  for (let i = 1; i < scrollEvents.length; i++) {
    if (scrollEvents[i].scrollY <= scrollEvents[i-1].scrollY) {
      return false; // Not strictly increasing
    }
  }
  
  return true;
}

async function issueSessionReceipt(session, sessionData) {
  const receiptData = {
    user: session.user_id,
    novel_id: session.novel_id,
    pages: [session.chapter_number],
    top_seconds: [Math.floor(sessionData.timeOnPage / 1000)],
    entropy_score: sessionData.entropyScore,
    device_hash: session.device_hash,
    timestamp: Math.floor(Date.now() / 1000)
  };
  
  // Sign with server Ed25519 private key
  const signature = signReceipt(receiptData);
  
  // Store receipt in database
  const { data: receipt } = await supabase
    .from('session_receipts')
    .insert({
      session_id: session.id,
      user_id: session.user_id,
      pages_read: receiptData.pages,
      time_on_pages: receiptData.top_seconds,
      entropy_score: receiptData.entropy_score,
      device_hash: receiptData.device_hash,
      server_signature: signature,
      expires_at: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString() // 24 hours
    })
    .select()
    .single();
  
  return {
    ...receiptData,
    sig: signature,
    receipt_id: receipt.id
  };
}

function signReceipt(data) {
  // In production, use proper Ed25519 signing
  const message = JSON.stringify(data);
  return crypto.createHash('sha256').update(message).digest('hex');
}

class QuestionGenerator {
  constructor() {
    this.openaiApiKey = process.env.OPENAI_API_KEY;
  }

  async generateQuestions(chapterText, chapterNumber, novelId) {
    // Check if questions already exist
    const existingQuestions = await this.getExistingQuestions(novelId, chapterNumber);
    if (existingQuestions.length > 0) {
      return existingQuestions;
    }

    const prompt = `
      Generate 4 multiple choice questions from this chapter text:
      ${chapterText.substring(0, 3000)} // First 3000 chars to avoid token limits
      
      Requirements:
      - Questions should test comprehension of key plot points
      - Each question must have exactly 4 options
      - Only one option should be correct
      - Questions should be answerable by someone who read the text
      - Avoid questions about minor details
      
      Format: JSON array with objects containing:
      - question: string (the question text)
      - options: array of 4 strings (multiple choice options)
      - correct_answer: integer (0-3, index of correct option)
    `;

    try {
      const response = await fetch('https://api.openai.com/v1/chat/completions', {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${this.openaiApiKey}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          model: "gpt-4",
          messages: [{ role: "user", content: prompt }],
          temperature: 0.7
        })
      });

      const data = await response.json();
      const questions = JSON.parse(data.choices[0].message.content);

      // Store questions in database
      await this.storeQuestions(novelId, chapterNumber, questions);

      return questions;
    } catch (error) {
      console.error('Question generation failed:', error);
      // Fallback to template questions
      return this.getFallbackQuestions();
    }
  }

  async getExistingQuestions(novelId, chapterNumber) {
    const { data } = await supabase
      .from('comprehension_challenges')
      .select('*')
      .eq('novel_id', novelId)
      .eq('chapter_number', chapterNumber);
    
    return data || [];
  }

  async storeQuestions(novelId, chapterNumber, questions) {
    for (const question of questions) {
      await supabase
        .from('comprehension_challenges')
        .insert({
          novel_id: novelId,
          chapter_number: chapterNumber,
          question: question.question,
          options: question.options,
          correct_answer: question.correct_answer,
          difficulty: 'medium'
        });
    }
  }

  getFallbackQuestions() {
    return [
      {
        question: "What was the main theme of this chapter?",
        options: ["Love and friendship", "Betrayal and revenge", "Adventure and discovery", "Mystery and suspense"],
        correct_answer: 0
      },
      {
        question: "How would you describe the pacing of this chapter?",
        options: ["Fast-paced and action-packed", "Slow and contemplative", "Varied pacing", "Consistent rhythm"],
        correct_answer: 2
      }
    ];
  }

  async getRandomChallenge(novelId, chapterNumber) {
    const questions = await this.getExistingQuestions(novelId, chapterNumber);
    if (questions.length === 0) {
      return null;
    }
    
    const randomIndex = Math.floor(Math.random() * questions.length);
    return questions[randomIndex];
  }
}

import React, { useState, useEffect } from 'react';

export default function ComprehensionChallenge({ challenge, onSubmit, onClose }) {
  const [selectedAnswer, setSelectedAnswer] = useState(null);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [timeRemaining, setTimeRemaining] = useState(60); // 60 seconds timer
  const [startTime] = useState(Date.now());

  useEffect(() => {
    const timer = setInterval(() => {
      setTimeRemaining(60 - Math.floor((Date.now() - startTime) / 1000));
    }, 1000);

    return () => clearInterval(timer);
  }, [startTime]);

  const handleSubmit = async () => {
    if (selectedAnswer === null) return;
    
    setIsSubmitting(true);
    const responseTime = Date.now() - startTime;
    
    await onSubmit(selectedAnswer, responseTime);
    setIsSubmitting(false);
  };

  const handleTimeout = () => {
    // Auto-submit with no answer on timeout
    onSubmit(-1, 60000); // -1 indicates timeout
  };

  if (timeRemaining <= 0) {
    handleTimeout();
    return null;
  }

  return (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50">
      <div className="bg-white rounded-lg p-6 max-w-md w-full mx-4">
        <div className="flex justify-between items-center mb-4">
          <h3 className="text-lg font-semibold">Reading Check</h3>
          <span className="text-sm text-gray-500">{timeRemaining}s</span>
        </div>
        
        <p className="text-gray-700 mb-6">{challenge.question}</p>
        
        <div className="space-y-3 mb-6">
          {challenge.options.map((option, index) => (
            <button
              key={index}
              onClick={() => setSelectedAnswer(index)}
              className={`w-full text-left p-4 rounded-lg border transition-all ${
                selectedAnswer === index 
                  ? 'border-blue-500 bg-blue-50 shadow-sm' 
                  : 'border-gray-300 hover:border-gray-400'
              }`}
            >
              <span className="font-medium">{String.fromCharCode(65 + index)}.</span> {option}
            </button>
          ))}
        </div>
        
        <div className="flex gap-3">
          <button
            onClick={handleSubmit}
            disabled={selectedAnswer === null || isSubmitting}
            className="flex-1 bg-blue-500 text-white py-3 rounded-lg font-medium disabled:bg-gray-300 disabled:cursor-not-allowed hover:bg-blue-600 transition-colors"
          >
            {isSubmitting ? 'Submitting...' : 'Submit Answer'}
          </button>
          
          <button
            onClick={onClose}
            className="px-4 py-3 border border-gray-300 rounded-lg hover:bg-gray-50 transition-colors"
          >
            Skip
          </button>
        </div>
        
        <p className="text-xs text-gray-500 mt-4 text-center">
          This helps us verify you're reading the content
        </p>
      </div>
    </div>
  );
}

export async function POST(request) {
  try {
    const { sessionId, challengeId, userAnswer, responseTime } = await request.json();
    
    // Get challenge details
    const challenge = await supabase
      .from('comprehension_challenges')
      .select('*')
      .eq('id', challengeId)
      .single();
    
    if (!challenge) {
      return NextResponse.json({ error: 'Invalid challenge' }, { status: 400 });
    }
    
    const isCorrect = userAnswer === challenge.correct_answer;
    
    // Record attempt
    await supabase
      .from('challenge_attempts')
      .insert({
        session_id: sessionId,
        challenge_id: challengeId,
        user_answer: userAnswer,
        is_correct: isCorrect,
        response_time_ms: responseTime
      });
    
    // Update user reputation
    await updateUserChallengeStats(session.user_id, isCorrect);
    
    // Check for consecutive failures
    if (!isCorrect) {
      await checkConsecutiveFailures(session.user_id);
    }
    
    return NextResponse.json({ 
      correct: isCorrect,
      canProceed: isCorrect || userAnswer === -1 // Allow proceed on timeout
    });
  } catch (error) {
    console.error('Challenge submission error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}

async function updateUserChallengeStats(userId, isCorrect) {
  const { data: reputation } = await supabase
    .from('reading_reputation')
    .select('*')
    .eq('user_id', userId)
    .single();
  
  if (!reputation) {
    // Create reputation record
    await supabase
      .from('reading_reputation')
      .insert({
        user_id: userId,
        challenges_attempted: 1,
        challenges_passed: isCorrect ? 1 : 0,
        consecutive_failed_challenges: isCorrect ? 0 : 1
      });
  } else {
    await supabase
      .from('reading_reputation')
      .update({
        challenges_attempted: reputation.challenges_attempted + 1,
        challenges_passed: reputation.challenges_passed + (isCorrect ? 1 : 0),
        consecutive_failed_challenges: isCorrect ? 0 : reputation.consecutive_failed_challenges + 1
      })
      .eq('user_id', userId);
  }
}

async function checkConsecutiveFailures(userId) {

      .update({ flagged_for_review: true })

      name: 'Seed',

    },
    reader: {

      cooldown: 48 * 60 * 60 * 1000, // 48 hours
      name: 'Reader',

      maxWithdrawal: 1000000, 
      cooldown: 24 * 60 * 60 * 1000, // 24 hours
      name: 'Scholar',
      description: 'Dedicated reader - Higher withdrawals'
    },
    sage: { 
      minScore: 1000, 
      maxWithdrawal: Infinity, 
      cooldown: 0, // No cooldown
      name: 'Sage',
      description: 'Master reader - Unlimited withdrawals'
    }
  };

  static async calculateRRS(userId) {
    // Get user's reading statistics
    const { data: reputation } = await supabase
      .from('reading_reputation')
      .select('*')
      .eq('user_id', userId)
      .single();
    
    if (!reputation) {
      return 0; // New user starts at 0
    }
    
    // Base score from pages read
    const pagesScore = reputation.total_pages_read * 1;
    
    // Bonus from challenge success rate
    const challengeSuccessRate = reputation.challenges_attempted > 0 
      ? reputation.challenges_passed / reputation.challenges_attempted 
      : 0;
    const challengeBonus = challengeSuccessRate * 200;
    
    // Bonus from total reading time (max 100 points)
    const readingTimeHours = reputation.total_reading_time_seconds / 3600;
    const timeBonus = Math.min(100, readingTimeHours * 2);
    
    // Consistency bonus based on regular reading patterns
    const consistencyBonus = await this.calculateConsistencyBonus(userId);
    
    // Penalty for flagged accounts
    const penalty = reputation.flagged_for_review ? -500 : 0;
    
    const totalScore = Math.max(0, pagesScore + challengeBonus + timeBonus + consistencyBonus + penalty);
    
    // Update the score in database
    await supabase
      .from('reading_reputation')
      .update({ 
        reputation_score: Math.floor(totalScore),
        current_tier: this.getTier(totalScore),
        updated_at: new Date().toISOString()
      })
      .eq('user_id', userId);
    
    return Math.floor(totalScore);
  }

  static async calculateConsistencyBonus(userId) {
    // Check if user has read in at least 3 of the last 7 days
    const sevenDaysAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000);
    
    const { data: recentSessions } = await supabase
      .from('reading_sessions')
      .select('created_at::date')
      .eq('user_id', userId)
      .gte('created_at', sevenDaysAgo.toISOString())
      .eq('status', 'completed');
    
    if (!recentSessions) return 0;
    
    const uniqueDays = new Set(recentSessions.map(s => s.created_at)).size;
    return uniqueDays >= 3 ? 50 : 0;
  }

  static getTier(score) {
    for (const [tierName, tierConfig] of Object.entries(this.TIERS)) {
      if (score >= tierConfig.minScore) return tierName;
    }
    return 'seed';
  }

  static async canWithdraw(userId, amount) {
    const reputation = await this.getUserReputation(userId);
    if (!reputation) return { canWithdraw: false, reason: 'No reputation data' };
    
    const tier = this.getTier(reputation.reputation_score);
    const tierConfig = this.TIERS[tier];
    
    // Check amount limit
    if (amount > tierConfig.maxWithdrawal) {
      return { 
        canWithdraw: false, 
        reason: `Amount exceeds tier limit of ${tierConfig.maxWithdrawal.toLocaleString()} SMP`,
        tier,
        tierConfig
      };
    }
    
    // Check cooldown
    if (reputation.last_withdrawal_at && tierConfig.cooldown > 0) {
      const timeSinceWithdrawal = Date.now() - new Date(reputation.last_withdrawal_at).getTime();
      if (timeSinceWithdrawal < tierConfig.cooldown) {
        const remainingTime = Math.ceil((tierConfig.cooldown - timeSinceWithdrawal) / (60 * 60 * 1000));
        return { 
          canWithdraw: false, 
          reason: `Cooldown active. ${remainingTime} hours remaining`,
          tier,
          tierConfig
        };
      }
    }
    
    return { canWithdraw: true, tier, tierConfig };
  }

  static async getUserReputation(userId) {
    const { data } = await supabase
      .from('reading_reputation')
      .select('*')
      .eq('user_id', userId)
      .single();
    
    return data;
  }

  static async updateReadingStats(userId, pagesRead, readingTimeSeconds) {
    const { data: reputation } = await supabase
      .from('reading_reputation')
      .select('*')
      .eq('user_id', userId)
      .single();
    
    if (!reputation) {
      // Create new reputation record
      await supabase
        .from('reading_reputation')
        .insert({
          user_id: userId,
          total_pages_read: pagesRead,
          total_reading_time_seconds: readingTimeSeconds,
          reputation_score: pagesRead,
          current_tier: this.getTier(pagesRead)
        });
    } else {
      // Update existing record
      const newPagesRead = reputation.total_pages_read + pagesRead;
      const newReadingTime = reputation.total_reading_time_seconds + readingTimeSeconds;
      const newScore = await this.calculateRRS(userId);
      
      await supabase
        .from('reading_reputation')
        .update({
          total_pages_read: newPagesRead,
          total_reading_time_seconds: newReadingTime,
          reputation_score: newScore,
          current_tier: this.getTier(newScore),
          updated_at: new Date().toISOString()
        })
        .eq('user_id', userId);
    }
  }

import { NextResponse } from 'next/server';
import { supabase } from '@/services/supabase/supabaseClient';
import { ReputationService } from '@/services/porp/ReputationService';

export async function POST(request) {
  try {
    const { userId, amount, receiptIds } = await request.json();
    
    // Validate withdrawal amount
    if (amount < 2500) {
      return NextResponse.json({ error: 'Minimum withdrawal is 2,500 SMP' }, { status: 400 });
    }
    
    // Check user balance
    const { data: balance } = await supabase
      .from('wallet_balances')
      .select('amount')
      .eq('user_id', userId)
      .eq('currency', 'SMP')
      .eq('chain', 'SOL')
      .single();
    
    if (!balance || balance.amount < amount) {
      return NextResponse.json({ error: 'Insufficient balance' }, { status: 400 });
    }
    
    // Check withdrawal eligibility based on RRS
    const eligibility = await ReputationService.canWithdraw(userId, amount);
    if (!eligibility.canWithdraw) {
      return NextResponse.json({ 
        error: eligibility.reason,
        tier: eligibility.tier,
        tierConfig: eligibility.tierConfig
      }, { status: 400 });
    }
    
    // Validate receipts if provided
    if (receiptIds && receiptIds.length > 0) {
      const receiptValidation = await validateReceipts(receiptIds, userId);
      if (!receiptValidation.valid) {
        return NextResponse.json({ error: 'Invalid or expired receipts' }, { status: 400 });
      }
    }
    
    // Create withdrawal request
    const { data: withdrawal } = await supabase
      .from('withdrawal_requests')
      .insert({
        user_id: userId,
        amount_smp: amount,
        tier_at_request: eligibility.tier,
        receipt_ids: receiptIds || [],
        cooldown_end_at: eligibility.tierConfig.cooldown > 0 
          ? new Date(Date.now() + eligibility.tierConfig.cooldown).toISOString()
          : null
      })
      .select()
      .single();
    
    // Mark receipts as used
    if (receiptIds && receiptIds.length > 0) {
      await supabase
        .from('session_receipts')
        .update({ used_for_claim: true })
        .in('id', receiptIds);
    }
    
    // Update last withdrawal time
    await supabase
      .from('reading_reputation')
      .update({ last_withdrawal_at: new Date().toISOString() })
      .eq('user_id', userId);
    
    return NextResponse.json({ 
      withdrawal: {
        id: withdrawal.id,
        amount: withdrawal.amount_smp,
        status: withdrawal.status,
        tier: eligibility.tier,
        cooldownEnds: withdrawal.cooldown_end_at
      }
    });
    
  } catch (error) {
    console.error('Withdrawal request error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}

async function validateReceipts(receiptIds, userId) {
  const { data: receipts } = await supabase
    .from('session_receipts')
    .select('*')
    .in('id', receiptIds)
    .eq('user_id', userId)
    .eq('used_for_claim', false);
  
  if (!receipts || receipts.length !== receiptIds.length) {
    return { valid: false };
  }
  
  // Check if any receipts are expired
  const now = new Date();
  const expiredReceipts = receipts.filter(r => new Date(r.expires_at) < now);
  
  if (expiredReceipts.length > 0) {
    return { valid: false };
  }
  
  return { valid: true, receipts };
}

export class SlashingService {
  static async flagCheater(userId, reason, severity = 'medium') {
    const stake = await this.getActiveStake(userId);
    if (!stake) return { slashed: false, reason: 'No active stake' };
    
    let slashPercentage = 0.3; // Default 30%
    
    if (severity === 'high') {
      slashPercentage = 0.5; // 50% for severe violations
    } else if (severity === 'low') {
      slashPercentage = 0.2; // 20% for minor violations
    }
    
    const slashAmount = stake.amount_smp * slashPercentage;
    
    // Apply slash
    await this.applySlash(userId, slashAmount, reason);
    
    return {
      slashed: true,
      amount: slashAmount,
      percentage: slashPercentage * 100,
      reason
    };
  }

  static async applySlash(userId, slashAmount, reason) {
    const stake = await this.getActiveStake(userId);
    
    // Burn portion (50% of slash)
    const burnAmount = slashAmount * 0.5;
    await this.burnTokens(burnAmount);
    
    // Redistribute portion (50% of slash)
    const redistributeAmount = slashAmount * 0.5;
    await this.redistributeToHonestUsers(redistributeAmount);
    
    // Update stake record
    await supabase
      .from('reputation_stakes')
      .update({
        is_active: false,
        slashed_at: new Date().toISOString(),
        slash_reason: reason
      })
      .eq('id', stake.id);
    
    // Log slash event
    await supabase
      .from('slash_events')
      .insert({
        user_id: userId,
        amount_slashed: slashAmount,
        amount_burned: burnAmount,
        amount_redistributed: redistributeAmount,
        reason: reason,
        created_at: new Date().toISOString()
      });
    
    // Update user reputation
    await supabase
      .from('reading_reputation')
      .update({
        flagged_for_review: true,
        reward_multiplier: 1.0,
        updated_at: new Date().toISOString()
      })
      .eq('user_id', userId);
  }

  static async getActiveStake(userId) {
    const { data } = await supabase
      .from('reputation_stakes')
      .select('*')
      .eq('user_id', userId)
      .eq('is_active', true)
      .single();
    
    return data;
  }

  static async burnTokens(amount) {
    // In a real implementation, this would send tokens to a burn address
    console.log(`Burning ${amount} SMP tokens`);
    
    // Track burn event
    await supabase
      .from('token_burns')
      .insert({
        amount: amount,
        reason: 'Stake slashing',
        created_at: new Date().toISOString()
      });
  }

  static async redistributeToHonestUsers(amount) {
    // Get honest users (not flagged, with active stakes)
    const { data: honestUsers } = await supabase
      .from('reading_reputation')
      .select('user_id')
      .eq('flagged_for_review', false)
      .gte('reputation_score', 100); // Reader tier and above
    
    if (!honestUsers || honestUsers.length === 0) return;
    
    // Distribute equally among honest users
    const amountPerUser = Math.floor(amount / honestUsers.length);
    
    for (const user of honestUsers) {
      await supabase
        .from('wallet_balances')
        .update({ 
          amount: supabase.rpc('amount', '+', amountPerUser)
        })
        .eq('user_id', user.user_id)
        .eq('currency', 'SMP')
        .eq('chain', 'SOL');
    }
    
    console.log(`Redistributed ${amount} SMP to ${honestUsers.length} honest users`);
  }

  static async getSlashEvents(userId, limit = 10) {
    const { data } = await supabase
      .from('slash_events')
      .select('*')
      .eq('user_id', userId)
      .order('created_at', { ascending: false })
      .limit(limit);
    
    return data || [];
  }
}

The 10-week implementation plan allows for gradual rollout and validation of each component, ensuring system stability and user acceptance. The modular architecture enables independent testing and deployment of each layer, reducing risk and allowing for optimization based on real-world usage.

Key benefits of this implementation:
- Eliminates bot drainage of reward pools
- Protects low-liquidity tokenomics
- Creates sustainable growth mechanics
- Enhances user engagement through gamification
- Provides transparent and fair reward distribution

The system is designed to be:
- Zero-friction for honest users
- Expensive and time-consuming for attackers
- Survivable at low liquidity levels
- Cryptographically verifiable and transparent
- Economically sustainable and self-regulating

With proper implementation and monitoring, PoRP will significantly enhance the security and economic stability of the SempaiHQ ecosystem while providing valuable insights into user behavior and engagement patterns.
